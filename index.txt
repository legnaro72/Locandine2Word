    <script>
        const BASE_WIDTH = 1200;
        const BASE_HEIGHT = 800;

        let scale = 1;
        let maxScale = 8;

        let posX = 0;
        let posY = 0;

        let pageFlip;

        const zoomLayer = document.getElementById("zoom-layer");
        const wrapper = document.getElementById("book-wrapper");
        const zoomIndicator = document.getElementById("zoom-indicator");

        /* ==== AUTO FIT INIZIALE ==== */
        function calculateInitialScale() {
            const scaleX = wrapper.clientWidth / BASE_WIDTH;
            const scaleY = wrapper.clientHeight / BASE_HEIGHT;
            scale = Math.min(scaleX, scaleY);
            centerContent();
        }

        function centerContent() {
            const scaledWidth = BASE_WIDTH * scale;
            const scaledHeight = BASE_HEIGHT * scale;

            posX = (wrapper.clientWidth - scaledWidth) / 2;
            posY = (wrapper.clientHeight - scaledHeight) / 2;
        }

        function clampPosition() {
            const scaledWidth = BASE_WIDTH * scale;
            const scaledHeight = BASE_HEIGHT * scale;

            const minX = wrapper.clientWidth - scaledWidth;
            const minY = wrapper.clientHeight - scaledHeight;

            if (scaledWidth > wrapper.clientWidth) {
                posX = Math.min(0, Math.max(minX, posX));
            } else {
                posX = (wrapper.clientWidth - scaledWidth) / 2;
            }

            if (scaledHeight > wrapper.clientHeight) {
                posY = Math.min(0, Math.max(minY, posY));
            } else {
                posY = (wrapper.clientHeight - scaledHeight) / 2;
            }
        }

        function updateTransform() {
            clampPosition();
            zoomLayer.style.transform =
                `translate(${posX}px, ${posY}px) scale(${scale})`;
            zoomIndicator.innerText = Math.round(scale * 100) + "%";
        }

        /* ==== ZOOM CONTROLS ==== */
        document.getElementById("zoom-in").onclick = (e) => {
            e.stopPropagation();
            scale += 0.2;
            updateTransform();
        };
        document.getElementById("zoom-out").onclick = (e) => {
            e.stopPropagation();
            scale = Math.max(scale - 0.2, 0.05);
            updateTransform();
        };
        document.getElementById("zoom-reset").onclick = (e) => {
            e.stopPropagation();
            calculateInitialScale();
            updateTransform();
        };

        function toggleFullScreen() {
            const btn = document.getElementById("fullscreen-toggle");
            if (!document.fullscreenElement) {
                document.documentElement.requestFullscreen().then(() => {
                    btn.innerText = "❌";
                    if (screen.orientation && screen.orientation.lock) {
                        screen.orientation.lock('landscape').catch(e => console.log("Orientation lock not supported", e));
                    }
                }).catch(err => {
                    console.error(`Error: ${err.message}`);
                });
            } else {
                if (document.exitFullscreen) {
                    document.exitFullscreen().then(() => {
                        btn.innerText = "⛶";
                    });
                }
            }
        }

        document.getElementById("fullscreen-toggle").onclick = (e) => {
            e.stopPropagation();
            toggleFullScreen();
        };

        // Aggiorna pulsante se si esce con ESC o tasti sistema
        document.addEventListener("fullscreenchange", () => {
            const btn = document.getElementById("fullscreen-toggle");
            if (!document.fullscreenElement) {
                btn.innerText = "⛶";
            } else {
                btn.innerText = "❌";
            }
        });

        /* ==== DOUBLE CLICK / TAP CON ZOOM CENTRATO ==== */
        let lastTap = 0;

        wrapper.addEventListener("pointerup", e => {
            if (dragOccurred) return;

            const now = Date.now();
            if (now - lastTap < 300) {

                const rect = wrapper.getBoundingClientRect();
                const clickX = e.clientX - rect.left;
                const clickY = e.clientY - rect.top;

                const offsetX = (clickX - posX) / scale;
                const offsetY = (clickY - posY) / scale;

                if (scale < 2) scale = 2;
                else if (scale < 4) scale = 4;
                else if (scale < 6) scale = 6;
                else if (scale < 8) scale = 8;
                else {
                    calculateInitialScale();
                    updateTransform();
                    lastTap = now;
                    return;
                }

                posX = clickX - offsetX * scale;
                posY = clickY - offsetY * scale;

                updateTransform();
            }
            lastTap = now;
        });

        /* ==== DRAG ==== */
        let isDragging = false;
        let dragOccurred = false;
        let startX, startY;

        wrapper.addEventListener("pointerdown", e => {
            isDragging = true;
            dragOccurred = false;
            startX = e.clientX - posX;
            startY = e.clientY - posY;
        });

        wrapper.addEventListener("pointermove", e => {
            if (isDragging) {
                const newX = e.clientX - startX;
                const newY = e.clientY - startY;

                if (Math.abs(newX - posX) > 5 || Math.abs(newY - posY) > 5) {
                    dragOccurred = true;
                }

                posX = newX;
                posY = newY;
                updateTransform();
            }
        });

        wrapper.addEventListener("pointerup", () => {
            isDragging = false;
        });

        /* ==== FRECCE ==== */
        document.getElementById("next").onclick = () => pageFlip.flipNext();
        document.getElementById("prev").onclick = () => pageFlip.flipPrev();

        /* ==== LOAD BOOK ==== */
        fetch('pages.json')
            .then(r => r.json())
            .then(pages => {

                const bookDiv = document.getElementById("book");

                pages.forEach(p => {
                    const div = document.createElement("div");
                    div.className = "page " + p.type;

                    const img = document.createElement("img");
                    img.src = p.image;

                    div.appendChild(img);
                    bookDiv.appendChild(div);
                });

                pageFlip = new St.PageFlip(bookDiv, {
                    width: 600,
                    height: 800,
                    size: "fixed",
                    showCover: true,
                    mobileScrollSupport: false,
                    useMouseEvents: false
                });

                pageFlip.loadFromHTML(document.querySelectorAll('.page'));

                calculateInitialScale();
                updateTransform();

                // Gestione Mobile e Fullscreen automatico
                const isMobile = /iPhone|iPad|iPod|Android/i.test(navigator.userAgent);
                const startOverlay = document.getElementById("mobile-start-overlay");

                if (isMobile) {
                    startOverlay.style.display = "flex";
                    document.getElementById("start-app").onclick = () => {
                        toggleFullScreen();
                        startOverlay.style.display = "none";
                    };
                }
            });

        window.addEventListener("resize", () => {
            // Non ricalcolare la scala automaticamente se l'utente ha zoomato manualimente
            // ma ricalcola se siamo alla scala iniziale (auto-fit)
            const oldInitialScale = Math.min(wrapper.clientWidth / BASE_WIDTH, wrapper.clientHeight / BASE_HEIGHT);
            if (Math.abs(scale - oldInitialScale) < 0.01) {
                calculateInitialScale();
            }
            updateTransform();
        });
    </script>
